"""
Project: Voice-Based Parkinson’s Disease Detection using CNN + BiLSTM
Author: Tushar (Hackathon Build)
Description:
This script builds a high-accuracy deep learning model for detecting Parkinson’s disease 
from voice recordings. The dataset consists of 5-second audio samples collected from 
patients and healthy controls. Each file may have an associated .txt description file 
with metadata. The data is stored in the "data" directory, containing subfolders for each class.

Datasets Referenced:
figdata:contains both healthy patient data in ./figdata/HC_AH
while for parkinsons there is ./figdata/PD_AH
also gitdata contains data of only diagnosed parkinsons patient

Goal:
Build a CNN + BiLSTM hybrid model that analyzes Mel-spectrograms of audio samples 
to classify between Parkinson’s and healthy controls with maximum accuracy.

Key Steps:  
1. Load and preprocess dataset from `data/` folder.
2. Extract Mel-spectrogram features from each audio file.
3. Build CNN + BiLSTM hybrid architecture.
4. Train with early stopping and learning rate scheduling.
5. Save model checkpoints and training history.
6. Evaluate on test set with precision, recall, F1-score.
7. Include a test script to run inference on new audio files.

Environment:
Python 3.10+
TensorFlow 2.15+
Librosa 0.10+
NumPy, Pandas, Scikit-learn, Matplotlib
"""

# =============================
# 📦 IMPORTS
# =============================
import os
import librosa
import numpy as np
import tensorflow as tf
from tensorflow.keras import layers, models
from sklearn.model_selection import train_test_split
from sklearn.metrics import classification_report, confusion_matrix
import matplotlib.pyplot as plt
import seaborn as sns

# =============================
# 📂 DATASET CONFIGURATION
# =============================
DATA_PATH = "data"
SAMPLE_RATE = 22050
DURATION = 5  # seconds
SAMPLES_PER_FILE = SAMPLE_RATE * DURATION

# =============================
# 🧹 DATA LOADER & FEATURE EXTRACTION
# =============================
def load_audio_files(data_path):
    """
    Loads audio files and corresponding labels from subdirectories.
    Assumes folder structure:
    data/
        parkinson/
            *.wav
        healthy/
            *.wav
    """
    X, y = [], []
    for label in os.listdir(data_path):
        label_path = os.path.join(data_path, label)
        if not os.path.isdir(label_path):
            continue
        for file in os.listdir(label_path):
            if file.endswith(".wav"):
                file_path = os.path.join(label_path, file)
                y.append(1 if "parkinson" in label.lower() else 0)
                X.append(file_path)
    return np.array(X), np.array(y)

def extract_melspectrogram(file_path, n_mels=128, n_fft=2048, hop_length=512):
    """Extracts a Mel-spectrogram from an audio file."""
    y, sr = librosa.load(file_path, sr=SAMPLE_RATE)
    if len(y) > SAMPLES_PER_FILE:
        y = y[:SAMPLES_PER_FILE]
    else:
        y = np.pad(y, (0, max(0, SAMPLES_PER_FILE - len(y))), "constant")
    mel = librosa.feature.melspectrogram(y=y, sr=sr, n_fft=n_fft, hop_length=hop_length, n_mels=n_mels)
    mel_db = librosa.power_to_db(mel, ref=np.max)
    return mel_db

def prepare_dataset(X, y):
    """Extracts features and converts to NumPy arrays suitable for CNN input."""
    features = []
    for i, path in enumerate(X):
        mel = extract_melspectrogram(path)
        features.append(mel)
        if i % 50 == 0:
            print(f"Processed {i}/{len(X)} files...")
    X = np.array(features)
    X = X[..., np.newaxis]  # add channel dimension
    return X, np.array(y)

# =============================
# 🧠 MODEL DEFINITION
# =============================
def build_cnn_bilstm(input_shape):
    """
    CNN + BiLSTM hybrid architecture:
    CNN extracts spectral patterns, BiLSTM captures temporal dependencies.
    """
    model = models.Sequential([
        layers.Conv2D(32, (3,3), activation='relu', padding='same', input_shape=input_shape),
        layers.MaxPooling2D((2,2)),
        layers.BatchNormalization(),

        layers.Conv2D(64, (3,3), activation='relu', padding='same'),
        layers.MaxPooling2D((2,2)),
        layers.BatchNormalization(),

        layers.Conv2D(128, (3,3), activation='relu', padding='same'),
        layers.MaxPooling2D((2,2)),
        layers.BatchNormalization(),

        layers.Reshape((-1, 128)),
        layers.Bidirectional(layers.LSTM(128, return_sequences=False)),
        layers.Dropout(0.4),
        layers.Dense(64, activation='relu'),
        layers.Dense(1, activation='sigmoid')
    ])

    model.compile(
        optimizer=tf.keras.optimizers.Adam(learning_rate=1e-4),
        loss='binary_crossentropy',
        metrics=['accuracy']
    )
    return model

# =============================
# ⚙️ TRAINING PIPELINE
# =============================
X_paths, y = load_audio_files(DATA_PATH)
X_train_paths, X_test_paths, y_train, y_test = train_test_split(X_paths, y, test_size=0.2, stratify=y, random_state=42)

print("Extracting features for training and test sets...")
X_train, y_train = prepare_dataset(X_train_paths, y_train)
X_test, y_test = prepare_dataset(X_test_paths, y_test)

input_shape = X_train.shape[1:]
model = build_cnn_bilstm(input_shape)

checkpoint = tf.keras.callbacks.ModelCheckpoint("best_model.h5", monitor="val_accuracy", save_best_only=True, mode="max")
early_stopping = tf.keras.callbacks.EarlyStopping(monitor="val_loss", patience=10, restore_best_weights=True)
lr_schedule = tf.keras.callbacks.ReduceLROnPlateau(monitor="val_loss", factor=0.5, patience=5, min_lr=1e-6)

history = model.fit(
    X_train, y_train,
    validation_split=0.2,
    epochs=100,
    batch_size=16,
    callbacks=[checkpoint, early_stopping, lr_schedule]
)

# =============================
# 📈 EVALUATION
# =============================
model.load_weights("best_model.h5")
y_pred = (model.predict(X_test) > 0.5).astype("int32")

print("Classification Report:\n", classification_report(y_test, y_pred))
cm = confusion_matrix(y_test, y_pred)

sns.heatmap(cm, annot=True, fmt='d', cmap='Blues', xticklabels=['Healthy', 'Parkinson'], yticklabels=['Healthy', 'Parkinson'])
plt.title("Confusion Matrix")
plt.xlabel("Predicted")
plt.ylabel("Actual")
plt.show()

# =============================
# 🧪 TEST SCRIPT (RUN INFERENCE)
# =============================
def predict_parkinson(file_path, model):
    """Predicts whether the input audio file indicates Parkinson’s disease."""
    mel = extract_melspectrogram(file_path)
    mel = mel[np.newaxis, ..., np.newaxis]
    prediction = model.predict(mel)[0][0]
    label = "Parkinson" if prediction > 0.5 else "Healthy"
    print(f"Prediction: {label} ({prediction*100:.2f}% confidence)")
    return label, prediction
